┌─────────────────────────────────────────────────────────────────────┐
│ TIPOS DE MENÚ                                                       │
├─────────────────────────────────────────────────────────────────────┤
│ SIMPLE                  │ Menú básico sin paginación                │
│ PAGINATION              │ Con paginación de items                   │
│ MULTI_PAGINATION        │ Múltiples secciones paginadas             │
│ FULL_INVENTORY          │ Acceso a inventario del jugador           │
│ PAGINATION_FULL         │ Paginación + inventario                   │
│ MULTI_PAGINATION_FULL   │ TODO combinado (más complejo)             │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ ESTRUCTURA BÁSICA                                                   │
├─────────────────────────────────────────────────────────────────────┤
│ title: '&6Título'       │ Título del menú                           │
│ type: SIMPLE            │ Tipo de menú                              │
│ size: 54                │ Tamaño (9, 18, 27, 36, 45, 54)            │
│                                                                     │
│ refresh:                                                            │
│   mode: SMART           │ DISABLED | FULL | SMART | SLOT_ONLY       │
│   interval: 20          │ Ticks (20 = 1 segundo)                    │
│                                                                     │
│ filler:                 │ Solo 2 tipos disponibles:                 │
│   global: {...}         │   - global (todo)                         │
│   border: {...}         │   - border (solo bordes)                  │
│                                                                     │
│ items:                  │ Items del menú                            │
│   item1: {...}                                                      │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ ITEM COMPLETO                                                       │
├─────────────────────────────────────────────────────────────────────┤
│ items:                                                              │
│   my_item:                                                          │
│     material: DIAMOND              # Material de Bukkit             │
│     amount: '1'                    # Cantidad                       │
│     name: '&bNombre'               # Nombre del item                │
│     lore: ['&7Línea 1']            # Lore (lista)                   │
│     slot: 13                       # Slot único                     │
│     slots: [10,11,12]              # Múltiples slots                │
│     slots: '10-16,19-25'           # Rangos                         │
│                                                                     │
│     enchantments:                  # Encantamientos                 │
│       DAMAGE_ALL: 5                                                 │
│                                                                     │
│     glowing: true                  # Brillo                         │
│     hide_attributes: true          # Ocultar atributos              │
│     unbreakable: true              # Irrompible                     │
│     max_stack_size: 1              # Stack máximo                   │
│     item_model: 1001               # Custom model data              │
│                                                                     │
│     potion: {type: SPEED}          # Configuración de poción        │
│     leather_armor: {color: '#FF0000'} # Color de armadura           │
│                                                                     │
│     actions:                 # Acciones al click              │
│       - 'message: &aHola'          # Sin prefijo = ANY              │
│       - 'left: close'              # Con prefijo = específico       │
│       - 'right: back'                                               │
│                                                                     │
│     dynamic_update: true           # Actualización dinámica         │
│     update_interval: 20            # Intervalo en ticks             │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ CLICK TYPES                                                         │
├─────────────────────────────────────────────────────────────────────┤
│ left         │ Click izquierdo                                      │
│ right        │ Click derecho                                        │
│ middle       │ Click del medio (rueda)                              │
│ shift_left   │ Shift + Click izquierdo                              │
│ shift_right  │ Shift + Click derecho                                │
│ drop         │ Tecla Q (drop)                                       │
│ swap         │ Tecla F (swap offhand)                               │
│ double       │ Doble click                                          │
│ number_key   │ Teclas 1-9                                           │
│ any          │ Cualquier tipo (default sin prefijo)                 │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ ACCIONES BUILT-IN                                                   │
├─────────────────────────────────────────────────────────────────────┤
│ close          │ Cierra el menú                                     │
│ back           │ Vuelve al menú anterior                            │
│ next_page      │ Siguiente página (PAGINATION)                      │
│ previous_page  │ Página anterior (PAGINATION)                       │
│ select: %index%│ Selecciona item (MULTI_PAGINATION)                 │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ PLACEHOLDERS                                                        │
├─────────────────────────────────────────────────────────────────────┤
│ GENERALES                                                           │
│   %menu_id%       │ UUID del menú                                   │
│   %menu_type%     │ Tipo de menú                                    │
│   %menu_title%    │ Título del menú                                 │
│   %player_name%   │ Nombre del jugador                              │
│                                                                     │
│ PAGINATION                                                          │
│   %current_page%  │ Página actual                                   │
│   %total_pages%   │ Total de páginas                                │
│   %index%         │ Índice global del item                          │
│   %page_index%    │ Índice en la página                             │
│                                                                     │
│ MULTI-PAGINATION                                                    │
│   %section_name%  │ Nombre de la sección                            │
│   %section_page%  │ Página de la sección                            │
│   %selected_index%│ Índice del item seleccionado                    │
│                                                                     │
│ CUSTOM (PlaceholderContext)                                         │
│   {custom_var}    │ Variables custom del contexto                   │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ PAGINATION                                                          │
├─────────────────────────────────────────────────────────────────────┤
│ pagination:                                                         │
│   slots: '10-16,19-25'   # Slots donde se muestran los items       │
│   items:                 # Lista de items a paginar                 │
│     - material: DIAMOND                                             │
│       name: 'Item #%index%'                                         │
│   navigation:            # Botones de navegación                    │
│     previous:                                                       │
│       slot: 48                                                      │
│       material: ARROW                                               │
│     next:                                                           │
│       slot: 50                                                      │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ MULTI-PAGINATION SECTIONS                                           │
├─────────────────────────────────────────────────────────────────────┤
│ sections:                                                           │
│   section1:                                                         │
│     name: 'Mi Sección'                                              │
│     slots: '10-16'                                                  │
│     filler: {...}        # Filler específico de sección             │
│     selected_template:   # Template para items seleccionados        │
│       glowing: true                                                 │
│       lore_append: ['&a✓ Seleccionado']                             │
│     items: [...]                                                    │
│     navigation:                                                     │
│       previous: {...}                                               │
│       next: {...}                                                   │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ SNAPSHOT + PLAYER INVENTORY                                         │
├─────────────────────────────────────────────────────────────────────┤
│ snapshot:                      # Solo en FULL_INVENTORY tipos       │
│   enabled: true                # Habilita snapshot                  │
│   restore_on_close: true       # Restaura al cerrar                 │
│   snapshot_id: 'menu_id'       # ID único                           │
│                                                                     │
│ player_inventory:              # Acceso al inventario               │

│     9-35  = Inventario principal                                    │

┌─────────────────────────────────────────────────────────────────────┐
│ QUICK TIPS                                                          │
├─────────────────────────────────────────────────────────────────────┤
│ • FILLER: Solo 'global' y 'border', nada más                        │
│ • SLOTS: Usa rangos '10-16' o listas [10,11,12] o mixto            │
│ • CLICK: Sin prefijo = ANY, con prefijo = específico                │
│ • REFRESH: Usa SMART para mejor performance                         │
│ • SNAPSHOT: Solo en tipos FULL_INVENTORY                            │
│ • PAGINATION: Usa navigation para prev/next                         │
│ • CUSTOM PATTERNS: Usa items normales, no hay más fillers           │
